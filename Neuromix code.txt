# server logic


server <- function(input, output, session) {
  
  output$no_results_message <- renderUI({
    tags$div(
      tags$p("Enter a gene name, or a list of genes separated by commas, and click 'Search NeurOmix Database'"),
      tags$p("Example Searches"),
      tags$p("HTT (mutations in Huntingtin (HTT) cause Huntington's Disease)"),
      tags$p("ATG101,ATG12,ATG13,ATG14,ATG16L1,ATG2A,ATG5,ATP13A2,ATP6V0A1 (search genes involved in autophagy)"), 
      tags$p("KIF1A,KIF1B,KIF1C,KIF21A,KIF3A,KIF3B,KIF4A,KIF5A,KIF5B,KIF5C (search genes involved in axonal transport)")
      
    )
    
  })
  
  #Info for user when clicking text above search buttons
  observeEvent(input$show_about_modal, {
    #Count columns, which equals number of gene lists
    x <- ncol(data)
    
    # Count unique cells in row 1, which is number of articles
    first_row <- as.character(unlist(data[1, ]))
    y <- length(unique(first_row))
    
    # Count non empty cells, which is number of genes
    z <- sum(!is.na(data) & data != "")
    
    modal_content <- switch(input$show_about_modal,
                            "NeurOmix" = HTML(paste("The NeurOmix database is a comprehensive manually curated collection of gene lists from high-throughput omics experiments. The studies included are broadly related to neuroscience and neurological diseases. 
                              <br> <br> Selecting <b>'Search NeurOmix'</b> scans the database for all gene lists containing the searched gene(s) and returns a summary of the experiments, as well as the rank of the identified gene and a link to the original article.
                              <br> <br> Selecting <b>'Search Co-Occurring Genes'</b> identifies all gene lists containing the searched gene and identifies genes that co-occur across gene lists.,
                              <br><br>The current version of NeurOmix includes", x, "gene lists from", y, "articles, containing", z, "gene entries")),
                            
                            
                            "interactome_data" = HTML(paste("Interactome data is drawn from API from BioGRID and STRING. BioGRID scores represent quantitative confidence scores as provided by the original publication, which vary by experimental methodology (e.g., CompPASS-Plus scores, SAINTexpress scores, or spectral count–based fold changes). STRING scores are computed from multiple lines of evidence (e.g., experimental, co-expression, text mining) resulting in a single likelihood score ranging from 0 to 1.")),
                            
                            
                            "coexpression" = HTML(paste(" .. "))
                            
    )
    
    showModal(modalDialog(
      title = "About NeurOmix",
      modal_content,
      easyClose = TRUE,
      footer = NULL
    ))
  })
  
  updateSelectInput(session, "column_choices", choices = as.character(colnames(data)))
  
  observe({
    updateTextInput(session, "textbox_genes", value = data[-(1:7), input$columns][data[-(1:7), input$columns] != ""])
  })
  
  observeEvent(input$keywords, {
    keyword <- input$keywords
    
    # If the user enters a keyword, create a regex pattern for filtering
    if (keyword != "") {
      keyword_pattern <- paste0("(?i)", keyword)  # (?i) makes the pattern case-insensitive
    } else {
      keyword_pattern <- ""  # No filtering if keyword is empty
    }
    
    # Filter columns containing the keyword
    filtered_columns <- colnames(data)[grepl(keyword_pattern, colnames(data))]
    
    # Update 'columns' selectInput choices with the filtered column names
    updateSelectInput(session, "columns", choices = c("", filtered_columns), selected = filtered_columns[1])
  })
  
  ################################################################## GENE SEARCH TAB
  
  
  # Get gene info from entrez
  
  #get_gene_details <- function(gene_name) {
  
  #   tryCatch({
  #    if (!gene_name %in% keys(org.Hs.eg.db, keytype = "SYMBOL")) {
  #     return(paste("Gene", gene_name, "not found."))
  #  }
  #      gene_id <- mapIds(
  #       org.Hs.eg.db, 
  #      keys = gene_name, 
  #     column = "ENTREZID", 
  #    keytype = "SYMBOL", 
  #   multiVals = "first"
  #      )
  #     if (is.na(gene_id)) {
  #       return(paste("Gene", gene_name, "not found."))
  #    }
  #   gene_summary <- entrez_summary(db = "gene", id = gene_id)
  #      paste(gene_name, ":", gene_summary$summary)
  #   }, error = function(e) paste("Error fetching details for", gene_name))
  #  }
  
  
  # Gene search observer
  observeEvent(input$searchgenes, {
    withProgress(message = "Searching NeurOmix. Analyzing gene lists may require several minutes to complete", {
      # Process input gene names
      gene_names <- toupper(trimws(unlist(strsplit(input$gene, ","))))
      
      if (length(gene_names) == 0) {
        output$gene_description <- renderUI({ NULL })
        output$search_results <- renderDT({ NULL })
        output$no_results_message <- renderUI({ p("Enter a gene name.") })
        return()
      }
      
      #   # Fetch gene descriptions
      #    output$gene_description <- renderUI({
      #      HTML(paste(sapply(gene_names, get_gene_details), collapse = "<br>"))
      #    })
      
      # Search for genes (with exact/partial matching based on checkbox)
      final_results <- do.call(rbind, lapply(gene_names, function(gene) {
        matches <- lapply(names(data_below_row_7), function(col) {
          if (input$exact_match) {
            indices <- which(data_below_row_7[[col]] == gene) # Exact match
          } else {
            indices <- which(grepl(gene, data_below_row_7[[col]], ignore.case = TRUE)) # Partial match
          }
          if (length(indices) > 0) {
            data.frame(
              Article = paste0('<a href="', data[7, col], '" target="_blank">', data[1, col], '</a>'),
              Description = paste0('<a href="#" class="experiment-summary" data-col="', col, '">', col, '</a>'),
              Gene = data_below_row_7[[col]][indices],
              Rank = indices,
              stringsAsFactors = FALSE
            )
          }
        })
        do.call(rbind, matches)
      }))
      
      # Render results
      if (!is.null(final_results) && nrow(final_results) > 0) {
        final_results <- final_results[!duplicated(final_results), ]
        final_results <- final_results[order(final_results$Rank), ]
        
        output$search_results <- renderDT({
          gene_text <- paste(gene_names, collapse = ", ")
          caption_text <- paste0(
            "<p><strong>NeurOmix Database Search Results</strong></p>",
            "<p>Queried genes: <em>", gene_text, "</em> were identified in ", nrow(final_results), " experiments. ",

  # Co-occurrence gene search observer
  observeEvent(input$cooccurrence, {
    withProgress(message = "Identifying co-occurring genes across NeurOmix...", {
      
      # Process input gene names
      gene_names <- toupper(trimws(unlist(strsplit(input$gene, ","))))
      gene_names <- gene_names[gene_names != ""]
      
      if (length(gene_names) == 0) {
        output$search_results     <- renderDT({ NULL })
        output$no_results_message <- renderUI({ p("Enter a gene name.") })
        return()
      }
      
      # -------------------------------
      # 1) Find experiments with seed genes
      # -------------------------------
      seed_hits <- do.call(rbind, lapply(gene_names, function(gene) {
        hits <- lapply(names(data_below_row_7), function(col) {
          col_vals <- data_below_row_7[[col]]
          
          if (input$exact_match) {
            # Exact, case-insensitive match
            indices <- which(toupper(col_vals) == gene)
          } else {
            # Partial match (case-insensitive)
            indices <- which(grepl(gene, col_vals, ignore.case = TRUE))
          }
          
          if (length(indices) > 0) {
            data.frame(
              ExperimentID = col,
              stringsAsFactors = FALSE
            )
          }
        })
        
        hits <- hits[!vapply(hits, is.null, logical(1))]
        if (length(hits) == 0) return(NULL)
        do.call(rbind, hits)
      }))
      
      if (is.null(seed_hits) || nrow(seed_hits) == 0) {
        output$search_results <- renderDT({
          datatable(
            data.frame(
              Message = "No NeurOmix experiments containing the searched gene(s) were found."
            ),
            rownames = FALSE,
            options = list(dom = 't')
          )
        })
        output$no_results_message <- renderUI({ NULL })
        return()
      }
      
      experiments_with_seed <- unique(seed_hits$ExperimentID)
      
      # -------------------------------
      # 2) Build gene/experiment pairs from those experiments
      # -------------------------------
      gene_exp_pairs <- do.call(rbind, lapply(experiments_with_seed, function(col) {
        genes_in_col <- data_below_row_7[[col]]
        genes_in_col <- toupper(trimws(genes_in_col))
        genes_in_col <- genes_in_col[genes_in_col != "" & !is.na(genes_in_col)]
        
        # Remove the seed genes; we only want co-occurring partners
        co_genes <- setdiff(unique(genes_in_col), gene_names)
        if (length(co_genes) == 0) return(NULL)
        
        data.frame(
          Gene         = co_genes,
          ExperimentID = col,
          stringsAsFactors = FALSE
        )
      }))
      
      if (is.null(gene_exp_pairs) || nrow(gene_exp_pairs) == 0) {
        output$search_results <- renderDT({
          datatable(
            data.frame(
              Message = "No co-occurring genes found for these search results."
            ),
            rownames = FALSE,
            options = list(dom = 't')
          )
        })
        output$no_results_message <- renderUI({ NULL })
        return()
      }
      
      # -------------------------------
      # 3) Summarize co-occurrence: experiments per gene
      # -------------------------------
      exp_counts <- aggregate(ExperimentID ~ Gene,
                              data = gene_exp_pairs,
                              FUN  = function(x) length(unique(x)))
      colnames(exp_counts)[2] <- "N_Experiments"
      
      exp_lists <- aggregate(ExperimentID ~ Gene,
                             data = gene_exp_pairs,
                             FUN  = function(x) paste(sort(unique(x)), collapse = "<br> "))
      colnames(exp_lists)[2] <- "Experiment_Lists"
      
      co_df <- merge(exp_counts, exp_lists, by = "Gene")
      
      # Keep only genes that appear in >= 2 seed-containing experiments
      co_df <- subset(co_df, N_Experiments >= 2)
      
      if (nrow(co_df) == 0) {
        output$search_results <- renderDT({
          datatable(
            data.frame(
              Message = "No co-occurring genes found (n ≥ 2) across NeurOmix gene lists."
            ),
            rownames = FALSE,
            options = list(dom = 't')
          )
        })
        output$no_results_message <- renderUI({ NULL })
        return()
      }
      
      # Order: most frequently co-occurring first
      co_df <- co_df[order(-co_df$N_Experiments, co_df$Gene), ]
      
      # -------------------------------
      # 4) Render as the main search_results table
      # -------------------------------
      output$search_results <- renderDT({
        caption_text <- paste0(
          "<p><strong>Co-occurring Genes Across NeurOmix Gene Lists</strong></p>",
          "<p>Queried gene(s): ", paste(gene_names, collapse = ", "), "</p>",
          "<p>These genes co-occur with the searched gene(s) in at least ",
          "2 NeurOmix gene lists. ",
          "<em>N_Experiments</em> indicates how many shared gene lists ",
          "each gene appears in, and <em>Experiment_Lists</em> lists those experiments (column IDs).</p>"
        )
        
        datatable(
          co_df,
          options = list(
            pageLength = 100,
            autoWidth  = TRUE,
            dom        = 'Bfrtip',
            buttons    = c('copy', 'csv', 'excel', 'pdf', 'print')
          ),
          escape = FALSE,
          rownames = FALSE,
          caption = tags$caption(
            style = 'caption-side: top; text-align: center; font-size: 14px; color: #000000;',
            HTML(caption_text)
          ),
          class = 'custom-datatable'
        )
      })
      
      output$no_results_message <- renderUI({ NULL })
    })
  })

  observeEvent(input$biogridstring, {  #combine BioGRID and STRING
    
    gene_name <- toupper(input$gene)
    
    merged_results <- data.frame("ProteinA" = character(),
                                 "ProteinB" = character(),
                                 "Score" = numeric(),
                                 "Source" = character(),
                                 stringsAsFactors = FALSE)
    
    # STRING Analysis
    if (gene_name != "") {
      mapped_protein <- tryCatch({
        rba_string_map_ids(gene_name, species = 9606, echo_query = FALSE, limit = NULL)
      }, error = function(e) NULL)  # Catch errors and set NULL
      
      if (!is.null(mapped_protein) && nrow(mapped_protein) > 0) {
        interaction_partners <- tryCatch({
          rba_string_interaction_partners(ids = mapped_protein$stringId,
                                          species = 9606,
                                          required_score = 300)
        }, error = function(e) NULL)
        
        if (!is.null(interaction_partners) && nrow(interaction_partners) > 0) {
          interaction_partners <- as.data.frame(interaction_partners)
          names(interaction_partners)[names(interaction_partners) == "preferredName_A"] <- "ProteinA"
          names(interaction_partners)[names(interaction_partners) == "preferredName_B"] <- "ProteinB"
          names(interaction_partners)[names(interaction_partners) == "escore"] <- "Score"
          
          interaction_partners <- interaction_partners[, c("ProteinA", "ProteinB", "Score")]
          interaction_partners$Source <- "STRING"
          
          # Append to merged_results
          merged_results <- rbind(merged_results, interaction_partners)
        } else {
          showNotification("No STRING interactions found for the given gene.", type = "warning")
        }
      } else {
        showNotification("Gene not found in STRING database. Please check the gene name.", type = "error")
      }
    }
    
    # BioGRID Analysis
    bg_get_key("AJ", "Keefe", "aj.keefe007612@gmail.com", "Neuromix")
    
    zB <- tryCatch({
      bg("interactions") %>%
        bg_constrain(geneList = gene_name) %>%
        bg_get_results()
    }, error = function(e) NULL)
    
    if (!is.null(zB) && is.data.frame(zB)) {
      if (all(c("organism_id_for_interactor_a", "quantitative_score") %in% colnames(zB))) {
        filtered_zB <- zB %>%
          dplyr::filter(
            organism_id_for_interactor_a %in% c("9606", "10090", "10116") &
              quantitative_score > 0.001
          ) %>%
          dplyr::select(8, 9, 19)
        
        if (nrow(filtered_zB) > 0) {
          names(filtered_zB) <- c("ProteinA", "ProteinB", "Score")
          filtered_zB$Source <- "BioGRID"
          
          # Append to merged_results
          merged_results <- rbind(merged_results, filtered_zB)
        } else {
          showNotification("No BioGRID interactions found for the given gene.", type = "warning")
        }
      }
    } else {
      showNotification("Error retrieving BioGRID data.", type = "error")
    }
    
    # Combined Results
    if (nrow(merged_results) > 0) {
      merged_results <- merged_results[order(merged_results$Score, decreasing = TRUE), ]
      
      output$interaction_results <- renderDT({
        datatable(merged_results,
                  options = list(pageLength = 50, autoWidth = TRUE),
                  rownames = FALSE,
                  class = 'interaction-table')
      })
    } else {
      output$interaction_results <- renderDT({
        datatable(data.frame(Message = "No interaction partners found. Is gene name correct?"),
                  options = list(pageLength = 1, autoWidth = TRUE),
                  rownames = FALSE)
      })
    }
  })

        x = "Score",

      ) +

  get_enriched_data <- function(selected_genes, db) {
    # Perform Enrichr analysis
    # Conditionally load enrichR only if not already loaded
    if (!requireNamespace("enrichR", quietly = TRUE)) {
      BiocManager::install("enrichR", update = FALSE, ask = FALSE)
    }
    library(enrichR)
    
    enriched <- as.data.frame(enrichr(selected_genes, db))
    
    # Remove underlines from columns
    colnames(enriched) <- gsub("[_.]", " ", colnames(enriched))
    
    # Round all the values to conserve space
    enriched[] <- lapply(enriched, function(x) {
      if (is.numeric(x)) round(x, 4) else x
    })
    
    # Remove unwanted columns
    columns_to_keep <- !grepl("Old | 2023 P value| Odds Ratio", colnames(enriched))
    
    # Subset the data frame to keep only the desired columns
    enriched <- enriched[, columns_to_keep]
    
    return(enriched)
  }

  observeEvent(input$interactome_analysis, {
    
    interactome_data <- fread("interactome_data.csv")

    selected_genes <- trimws(toupper(selected_genes)) # Convert to lowercase and remove any leading/trailing whitespace
    
    # Convert ProteinA and ProteinB to lowercase
    interactome_data <- interactome_data %>%
      mutate(ProteinA = toupper(ProteinA), ProteinB = toupper(ProteinB))
    
    # Find PPIs for all proteins from the selected gene list
    interactome1 <- interactome_data %>%
      filter(ProteinA %in% selected_genes | ProteinB %in% selected_genes) %>%
      dplyr::select(ProteinA, ProteinB)
    
    # Create a combined data frame of interactions
    interactome1 <- as.data.frame(cbind(interactome1$ProteinA, interactome1$ProteinB))
    colnames(interactome1) <- c("ProteinA", "ProteinB")
    
    # Count the interactions for each protein in the list
    interactome1 <- interactome1 %>%
      mutate(Co_Interactor = as.character(ifelse(ProteinA %in% selected_genes, ProteinB, ProteinA))) %>%
      filter(!Co_Interactor %in% selected_genes) %>%
      group_by(Co_Interactor) %>%
      summarize(Interacting.Protein.Count = n_distinct(ifelse(ProteinA %in% selected_genes, ProteinA, ProteinB)),
                Interacting.Proteins.From.Gene.List = paste(unique(ifelse(ProteinA %in% selected_genes, ProteinA, ProteinB)), collapse = ", ")) %>%
      arrange(desc(Interacting.Protein.Count))
    
    # Calculate the total number of interactions for each Co_Interactor across the entire interactome_data dataset
    total_interactions <- interactome_data %>%
      mutate(Co_Interactor = ifelse(ProteinA %in% selected_genes, ProteinB, ProteinA)) %>%
      group_by(Co_Interactor) %>%
      summarize(Total.Interaction.Count.Of.Co_Interactor = n()) %>%
      ungroup()
    
    # Merge the two data frames to include the total number of interactions
    interactome1 <- interactome1 %>%
      left_join(total_interactions, by = "Co_Interactor") %>%
      mutate(Percent.Of.Co_Interactor.Interactions.in.Gene.List = round((Interacting.Protein.Count / Total.Interaction.Count.Of.Co_Interactor * 100),3)) %>% 
      filter(Interacting.Protein.Count > 2) %>%
      arrange(desc(Percent.Of.Co_Interactor.Interactions.in.Gene.List)) %>% 
      rename_with(~ gsub("\\.", " ", .)) 
    
    output$gofunction_results <- renderDT({
      datatable(interactome1, class = 'custom-datatable', options = list(pageLength = 50, autoWidth = TRUE, columnDefs = list(list(width = '200px', targets = c(1, 3)))
      ), rownames = FALSE)
    })
    
    output$gofunction_graph <- renderPlot({
      ggplot(interactome1 %>% slice_head(n = 20), aes(x = reorder(Co_Interactor, `Percent Of Co_Interactor Interactions in Gene List`), y = `Percent Of Co_Interactor Interactions in Gene List`)) +
        geom_point(color = "blue", size = 3) +
        geom_text(aes(label = `Total Interaction Count Of Co_Interactor`), vjust = -.7, size = 4) +
        labs(
          x = "Co-Interacting Protein (Top 20 Hits Displayed)",
          y = "Percent Of Co_Interactor's Interactions in Gene List"
        ) +
        ggtitle("Percent of co interactor's interactions identified in gene list. The total number of co interactor interactions in text") +  # Add the title here
        
        theme(
          strip.background = element_rect(fill = "darkblue"), 
          strip.text = element_text(size = 12, colour = "white"),
          panel.border = element_rect(color = "black", fill = NA, size = .5),  # Add border to the panel
          panel.background = element_rect(fill = "#e4ebf0"),
          plot.background = element_rect(fill = "#e4ebf0", color = "black", size = 1),  # Add border to the entire plot
          axis.text.x = element_text(size = 10, angle = 20, vjust = .7),
          axis.text.y = element_text(size = 10, angle = 90),
